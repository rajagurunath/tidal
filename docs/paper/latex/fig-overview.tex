  {\textbf{Tidal gateway \; (placement A)}\\[2pt]

   \circled{2} durable store (\code{/v1/files}$+$\code{/v1/batches})\\
   \circled{3} claim in deadline order at laxity priority\\
   \circled{4} AIMD throttle at 1\,Hz on \code{/metrics}\\
   \end{tabular}};

% engine (with placement B inside)
\node[major, right=17mm of gw] (engine)
  {\textbf{unmodified vLLM}\\[1.5pt]
   priority scheduling, chunked prefill, budget $\tau$\\[2.5pt]
   \begin{tikzpicture}[inner sep=3pt]
     \node[box, draw=accent!70!black, dash pattern=on 2.4pt off 1.8pt, align=left]

   \end{tikzpicture}};

  {online users};

\caption[System overview: one policy, two placements]{%
  \textbf{One policy, two placements.} Batch jobs enter through an OpenAI-wire gateway
  that \circled{1} refuses infeasible windows at admission, \circled{2} stores jobs
  durably, \circled{3} claims items in deadline order at their laxity priority, and
  \circled{4} throttles dispatch by AIMD over the engine's public metrics. Placement A
  stops there: the engine is stock vLLM, and online traffic never traverses the gateway.
  Placement B moves admission inside the engine \circled{5} as a scheduler plugin that
  caps batch tokens per iteration against a self-calibrated latency model. Both are
  plugins; neither forks the engine.}
