\begin{figure}[t]

\begin{tikzpicture}[

  ptr/.style={-{Stealth[length=1.5mm,width=1.3mm]}, draw=black!55, line width=0.4pt},
]

% floor region: urgency > 0.9 for the at-risk job => t in (19.4, 20)..deadline
\fill[accent!14] (19.4,0) rectangle (24,112);

% axes

% y ticks
\draw[black!70] (-0.8,0)   -- (-1.04,0)   node[left, inner sep=2pt, yshift=-4pt] {0};
\draw[black!70] (-0.8,1)   -- (-1.04,1)   node[left, inner sep=2pt, yshift=6pt]  {1};

% axis labels
\node[ann, anchor=north] at (12,-16) {wall-clock time since submission};

\draw[accent, line width=1.2pt] (0,100) -- (24,100);

\caption[Three escalation policies over one 24-hour window]{%
  \textbf{The contract in one picture.} Wall-clock aging (gray dashes) escalates every
  job on the clock alone. Laxity escalation separates the cases: an on-track job (solid)
  stays at the lowest band for the entire window, while an at-risk job (dotted) ramps
  linearly over its last $H = 6$\,h of \emph{slack} --- beginning when its laxity crosses
  $H$, not at any fixed hour --- to one notch below online. Above urgency $0.9$ (shaded) a
  token bucket guarantees a minimum admission rate, whose assured progress feeds back into
  laxity and stabilizes the job at the floor; \texttt{sla\_strict} allows the dashed final
  step to priority $0$ --- parity --- exactly at the deadline.}
\label{fig:priority}
\end{figure}
